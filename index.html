function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var patSheet = ss.getSheetByName("patients") || ss.getSheets()[0]; var appSheet = ss.getSheetByName("appointments") || ss.getSheets()[1]; var patData = sheetToObjects(patSheet); var appData = sheetToObjects(appSheet); return ContentService.createTextOutput(JSON.stringify({ patients: patData, appointments: appData })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // אם זו בקשה להפקת חשבונית אמיתית במורנינג if (data.action === "createMorningDoc") { var result = createMorningDocument(data); return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } // שמירת כל הנתונים מחדש בגיליון if (data.patients) saveToSheet(ss.getSheetByName("patients") || ss.getSheets()[0], data.patients); if (data.appointments) saveToSheet(ss.getSheetByName("appointments") || ss.getSheets()[1], data.appointments); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // פונקציית פנייה אמיתית ל-API של Morning function createMorningDocument(req) { var apiKey = req.apiKey; var apiSecret = req.apiSecret; if (!apiKey || !apiSecret) { return { success: false, message: "חווטי API של Morning חסרים" }; } // 1. קבלת Token ממורנינג var tokenUrl = "https://api.greeninvoice.co.il/v1/account/token"; var tokenOptions = { method: "post", contentType: "application/json", payload: JSON.stringify({ id: apiKey, secret: apiSecret }), muteHttpExceptions: true }; var tokenRes = UrlFetchApp.fetch(tokenUrl, tokenOptions); var tokenJson = JSON.parse(tokenRes.getContentText()); if (!tokenJson.token) { return { success: false, message: "שגיאת אימות מול Morning: " + (tokenJson.message || "מפתח API שגוי") }; } var jwtToken = tokenJson.token; // 2. יצירת חשבונית מס קבלה (סוג מסמך 320) var docUrl = "https://api.greeninvoice.co.il/v1/documents"; // ניקוי מקפים מהטלפון לזיהוי רציף var cleanPhone = String(req.clientPhone || "").replace(/[^0-9]/g, ""); var payTypeMap = { "מזומן": 1, "צ'ק": 2, "כרטיס אשראי": 3, "העברה בנקאית": 4, "ביט / פייבוקס": 10 }; var payload = { type: 320, // 320 = חשבונית מס קבלה client: { name: req.clientName, phone: cleanPhone }, income: [ { description: req.description || "טיפול/ים בקליניקה", quantity: 1, price: Number(req.amount), vatType: 1 // 1 = מחיר כולל מע"מ (עוסק מורשה) } ], payment: [ { type: payTypeMap[req.paymentMethod] || 10, price: Number(req.amount) } ] }; var docOptions = { method: "post", contentType: "application/json", headers: { Authorization: "Bearer " + jwtToken }, payload: JSON.stringify(payload), muteHttpExceptions: true }; var docRes = UrlFetchApp.fetch(docUrl, docOptions); var docJson = JSON.parse(docRes.getContentText()); if (docJson.id) { return { success: true, docNumber: docJson.number, pdfUrl: docJson.url ? docJson.url.he : "" }; } else { return { success: false, message: docJson.message || "שגיאה ביצירת המסמך במורנינג" }; } } function sheetToObjects(sheet) { if (!sheet) return []; var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var headers = values[0]; var result = []; for (var i = 1; i < values.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = values[i][j]; } result.push(obj); } return result; } function saveToSheet(sheet, data) { if (!sheet || !data || data.length === 0) return; sheet.clear(); var headers = Object.keys(data[0]); sheet.appendRow(headers); data.forEach(function(row) { var arr = headers.map(function(h) { return row[h] !== undefined ? row[h] : ""; }); sheet.appendRow(arr); }); }
